Любая 2D игра работает по принципу непрерывного игрового цикла: Очистка кадра ➔ Обновление координат ➔ Перерисовка объектов. Функция requestAnimationFrame синхронизирует обновления с частотой развёртки монитора (60 FPS).

Схема системы координат HTML5 Canvas (600 × 700)
Код игрового цикла 
const canvas = document.getElementById('gameCanvas');
const ctx = canvas.getContext('2d');

let lastTime = 0;

function gameLoop(timestamp) {
  const deltaTime = timestamp - lastTime;
  lastTime = timestamp;

  // 1. Очищаем холст
  ctx.clearRect(0, 0, canvas.width, canvas.height);

  // 2. Обновляем и рисуем объекты
  update(deltaTime);
  render();

  // 3. Запрашиваем следующий кадр
  requestAnimationFrame(gameLoop);
}

requestAnimationFrame(gameLoop);

Для плавной обработки одновременно зажатых клавиш удобно использовать структуру new Set(). Чтобы корабль не вылетал за границы Canvas, мы удерживаем позицию методом Math.max() и Math.min().

Класс Ship и обработчик клавиатуры 
const keysPressed = new Set();
window.addEventListener('keydown', (e) => keysPressed.add(e.code));
window.addEventListener('keyup', (e) => keysPressed.delete(e.code));

class Ship {
  constructor(x, y) {
    this.x = x;
    this.y = y;
    this.radius = 18;
    this.speed = 7;
  }

  update(canvasWidth) {
    if (keysPressed.has('ArrowLeft') || keysPressed.has('KeyA')) this.x -= this.speed;
    if (keysPressed.has('ArrowRight') || keysPressed.has('KeyD')) this.x += this.speed;

    // Ограничение движения в пределах экрана
    this.x = Math.max(this.radius, Math.min(canvasWidth - this.radius, this.x));
  }

  draw(ctx) {
    ctx.save();
    ctx.translate(this.x, this.y);
    ctx.beginPath();
    ctx.moveTo(0, -22); // Нос
    ctx.lineTo(18, 22); // Правое крыло
    ctx.lineTo(-18, 22); // Левое крыло
    ctx.closePath();
    ctx.strokeStyle = '#00f0ff';
    ctx.lineWidth = 2;
    ctx.stroke();
    ctx.restore();
  }
}

Math.cos() / Math.sin() ctx.translate() / rotate() Asteroid.split()

Астероиды создаются процедурно без сторонних картинка-файлов. Мы рассчитываем координаты каждой неровной вершины многоугольника с помощью тригонометрии в полярных координатах: x = r * cos(θ) и y = r * sin(θ).

Тригонометрическое построение вершин астероида
Генерация вершин и раскалывание 
class Asteroid {
  constructor(x, y, radius = 30, level = 3) {
    this.x = x;
    this.y = y;
    this.radius = radius;
    this.level = level; // Level 3: Крупный, 2: Средний, 1: Малый
    this.points = [];
    
    // Генерация 8-12 вершин со случайным отклонением радиуса
    const verts = 9;
    for (let i = 0; i < verts; i++) {
      const angle = (i / verts) * Math.PI * 2;
      const r = this.radius * (0.75 + Math.random() * 0.5);
      this.points.push({ x: Math.cos(angle) * r, y: Math.sin(angle) * r });
    }
  }

  split() {
    if (this.level <= 1) return [];
    // При раскалывании создаём два меньших осколка
    return [
      new Asteroid(this.x - 10, this.y, this.radius * 0.6, this.level - 1),
      new Asteroid(this.x + 10, this.y, this.radius * 0.6, this.level - 1)
    ];
  }
}
Math.hypot(dx, dy) Радиальное пересечение

Для мгновенной проверки столкновения двух круглых объектов в 2D используется формула евклидова расстояния: d = √((x₂ - x₁)² + (y₂ - y₁)²). Современный JS предоставляет эффективный метод Math.hypot(dx, dy).

Геометрия детекции столкновений (Hit-test)
Столкновение ЕСТЬ, если: d < (R₁ + R₂)
Функция проверки коллизии 
function checkCollision(obj1, obj2) {
  const dx = obj1.x - obj2.x;
  const dy = obj1.y - obj2.y;
  // Math.hypot высчитывает гипотенузу: Math.sqrt(dx*dx + dy*dy)
  const distance = Math.hypot(dx, dy);
  return distance < (obj1.radius + obj2.radius);
}
class Particle ctx.globalAlpha Screen Shake Offset

Взрывы создаются разлётом мелких частиц с затуханием прозрачности globalAlpha. Тряска экрана реализуется смещением всего холста на случайные значения ctx.translate(shakeX, shakeY) при получении урона.

Система частиц и Тряска экрана 
class Particle {
  constructor(x, y, color) {
    this.x = x;
    this.y = y;
    this.color = color;
    this.radius = Math.random() * 3 + 1;
    const angle = Math.random() * Math.PI * 2;
    const speed = Math.random() * 5 + 1;
    this.vx = Math.cos(angle) * speed;
    this.vy = Math.sin(angle) * speed;
    this.alpha = 1;
  }

  update() {
    this.x += this.vx;
    this.y += this.vy;
    this.alpha -= 0.02; // Постепенное затухание
  }

  draw(ctx) {
    ctx.save();
    ctx.globalAlpha = Math.max(0, this.alpha);
    ctx.fillStyle = this.color;
    ctx.beginPath();
    ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
    ctx.fill();
    ctx.restore();
  }
}
new AudioContext() createOscillator() localStorage.setItem()

Для максимальной производительности и независимости от внешних файл-ресурсов выстрелы и взрывы генерируются на лету через **Web Audio API** с помощью генераторов частот (OscillatorNode). Рекорд сохраняется в браузере пользователя через localStorage.

Синтез выстрела лазера и Сохранение 
const audioCtx = new (window.AudioContext || window.webkitAudioContext)();

function playLaserSound() {
  const osc = audioCtx.createOscillator();
  const gain = audioCtx.createGain();
  osc.type = 'sawtooth';
  
  // Быстрое падение частоты от 800 Гц до 100 Гц (эффект пиу!)
  osc.frequency.setValueAtTime(800, audioCtx.currentTime);
  osc.frequency.exponentialRampToValueAtTime(100, audioCtx.currentTime + 0.1);
  
  gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
  gain.gain.linearRampToValueAtTime(0.01, audioCtx.currentTime + 0.1);
  
  osc.connect(gain);
  gain.connect(audioCtx.destination);
  osc.start();
  osc.stop(audioCtx.currentTime + 0.1);
}

// Настройка звуков через Web Audio API
const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
function playSound(freq, duration, type = 'sine') {
  if (audioCtx.state === 'suspended') audioCtx.resume();
  const osc = audioCtx.createOscillator();
  const gain = audioCtx.createGain();
  osc.type = type;
  osc.frequency.setValueAtTime(freq, audioCtx.currentTime);
  gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
  gain.gain.linearRampToValueAtTime(0.01, audioCtx.currentTime + duration);
  osc.connect(gain);
  gain.connect(audioCtx.destination);
  osc.start();
  osc.stop(audioCtx.currentTime + duration);
}

// Классы снарядов и астероидов
class Bullet {
  constructor(x, y) {
    this.x = x;
    this.y = y;
    this.radius = 4;
    this.speed = 12;
  }
  update() { this.y -= this.speed; }
  draw() {
    ctx.beginPath();
    ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
    ctx.fillStyle = '#ff007f';
    ctx.fill();
  }
}

class Asteroid {
  constructor(x, y, radius = 30) {
    this.x = x || Math.random() * (canvas.width - 60) + 30;
    this.y = y || -40;
    this.radius = radius;
    this.speed = Math.random() * 2 + 1.5;
  }
  update() { this.y += this.speed; }
  draw() {
    ctx.beginPath();
    ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
    ctx.strokeStyle = '#94a3b8';
    ctx.lineWidth = 2;
    ctx.stroke();

let player = new Ship();
let bullets = [];
let asteroids = [];

function startGame() {
  player = new Ship();
  bullets = [];
  asteroids = [];
  score = 0;
  gameState = 'playing';
  document.getElementById('overlay').style.display = 'none';
}

function gameLoop() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);

  if (gameState === 'playing') {
    player.update();
    player.draw();

    if (keys['Space']) {
      if (!player.lastShoot || Date.now() - player.lastShoot > 200) {
        bullets.push(new Bullet(player.x, player.y - 20));
        playSound(600, 0.1, 'sawtooth');
        player.lastShoot = Date.now();
      }
    }

    if (Math.random() < 0.02) asteroids.push(new Asteroid());

    bullets.forEach(b => { b.update(); b.draw(); });
    asteroids.forEach(a => { a.update(); a.draw(); });

    // Проверка столкновений
    bullets.forEach((b, bi) => {
      asteroids.forEach((a, ai) => {
        if (Math.hypot(b.x - a.x, b.y - a.y) < b.radius + a.radius) {
          bullets.splice(bi, 1);
          asteroids.splice(ai, 1);
          score += 10;
          document.getElementById('score').textContent = score;
          playSound(150, 0.2, 'square');
        }
      });
    });

    bullets = bullets.filter(b => b.y > 0);
    asteroids = asteroids.filter(a => a.y < canvas.height + 50);
  }

  requestAnimationFrame(gameLoop);
}

requestAnimationFrame(gameLoop);
